\section*{Termination of the meet computation}

\todoin{\textbf{Proof overview.}
	The main difficulty is to show that the arguments are drawn from a finite set.
	This is clear if no variable expansion occurs:
	every recursive call then receives either a subterm of a current argument
	or a definition body from $S$.
	In this case, the call guard set prevents infinite unfolding of recursive types.
	Variable expansion is the only case that introduces expressions from outside this set.
	Assuming that at least one of the initial type-expression arguments
	and the initial mapping are variable-free,
	at most one variable expansion can occur along branches.
	After that expansion, both arguments are variable-free and remain so.
	Hence termination reduces to the variable-free case.}

\subsection*{Definitions}

For a type expression $e$, the set $\subexp(e)$ of its \emph{subterms} is defined by
\[
	\subexp(e) = \{e\} \cup
	\begin{cases}
		\bigcup_{i=1}^n \subexp(e_i) & \text{if } e = f(e_1,\dots,e_n) \text{ or } e = e_1 \mid \dots \mid e_n, \\
		\emptyset                    & \text{otherwise.}
	\end{cases}
\]


Let $S_{\mathrm{sub}} = \bigcup_{(t \lessdot e) \in S\ \text{or}\ (t \doteq e) \in S} \subexp(e)$
be the set of subterms of all definition bodies. It is finite.

Let $T_S = \{\, t : (t \lessdot e) \in S \text{ or } (t \doteq e) \in S \,\}$
be the constants defined in $S$. $T_S$ is finite, since $S$ is.

For a type expression $e$, its \emph{size} is defined by
\[
	\mathrm{size}(e) =
	\begin{cases}
		1 + \sum_{i} \mathrm{size}(e_i) & \text{if } e = f(e_1,\dots,e_n) \text{ or } e = e_1 \mid \dots \mid e_n \\
		1                               & \text{otherwise.}
	\end{cases}
\]

An expression is \emph{variable-free} if no type variable occurs in it.
A mapping $\sigma \in \Sigma$ is \emph{variable-free} if $\sigma(v)$ is variable-free for every $v \in V$.

For reference in the termination proof, we name the cases of $\meet_S$ in the order they are defined:
\emph{top}, \emph{type unfolding}, \emph{variable expansion}, \emph{union decomposition},
\emph{function type}, \emph{subtype unfolding}, \emph{function decomposition},
\emph{constants}, and \emph{fall-through}.
The \emph{recursive calls} of a case are the applications of $\meet_S$ occurring on its right-hand side.
The cases \emph{top}, \emph{function type}, \emph{constants}, and \emph{fall-through} make no recursive call,
and we refer to them collectively as the \emph{terminal cases}.

The \emph{call tree} of $\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$ has that call as its root,
and at each node the recursive calls made by the applicable case as its children.
We identify termination of $\meet_S$ with finiteness of its call tree.
A \emph{var-step} is a step using variable expansion.

\subsection*{Lemmas}

\begin{lemma}[variable-freeness invariant]
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	and $C \subseteq T \times \mathcal{T}$.

	If type expressions $e_1$ and $e_2$ are variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	both $e_1'$ and $e_2'$ are variable-free.
\end{lemma}

\begin{proof}
	By inspection of the definition of $\meet_S$,
	variable-freeness of the first two arguments is preserved under direct recursive subcalls.
	Since the initial call $\meet_S(e_1,e_2,\sigma,C)$ has variable-free
	first two arguments, it follows that every reachable subcall
	$\meet_S(e_1',e_2',\sigma',C')$ also has variable-free first two
	arguments.
\end{proof}

If the type expressions in all recursive calls are variable-free,
we obtain the following corollary.

\begin{corollary}
	There is no variable expansion step during the evaluation of a meet with
	two variable-free type expressions.
\end{corollary}

The next invariant shows that we can restrict attention to a finite set of type
expressions when analyzing the meet computation.

\begin{lemma}[finite arguments invariant]
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	$C \subseteq T \times \mathcal{T}$,
	and $P = \subexp(e_1) \cup \subexp(e_2) \cup \subexp(S)$.

	If type expressions $e_1$ and $e_2$ are variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	$e_1',e_2' \in P$.
\end{lemma}

\begin{proof}
	Since there is no variable expansion step during the evaluation of a meet with
	two variable-free type expressions, every subcall receives either a subterm of
	one of the current arguments or a definition body from $S$. Therefore, all
	arguments of all reachable subcalls are contained in the finite set $P$.
\end{proof}

\begin{lemma}
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	and $C \subseteq T \times \mathcal{T}$.

	If type expressions $e_1$ and $e_2$ are variable-free,
	then the meet computation $\meet_S(e_1,e_2,\sigma,C)$ terminates.
\end{lemma}

\begin{proof}
	Proof idea: we are either increasing the call guard set $C$ or decreasing
	the size of the type expression arguments. Since both are finite, we cannot
	have an infinite sequence of recursive calls.
\end{proof}

\begin{lemma}[one-sided variable-freeness invariant]
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	and $C \subseteq T \times \mathcal{T}$.

	If at least one of the expressions $e_1$ and $e_2$ is variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	at least one of the expressions $e_1'$ and $e_2'$ is variable-free.
\end{lemma}

\begin{proof}
	By inspection of the definition of $\meet_S$, it suffices to show that
	the property of having at least one variable-free expression among the
	first two arguments is preserved under direct recursive subcalls.

	This is immediate for all recursive cases except variable expansion,
	where a substitution is applied to a variable and the meet is computed
	between the substituted expression and the other argument:
	$\meet_S(v,e,\sigma,C)$.
	Since at least one of the two arguments of the current call is
	variable-free, the expression $e$ must be variable-free. Hence, the
	subsequent call $\meet_S(\sigma(v),e,\sigma,C)$ preserves the invariant.
\end{proof}

\begin{lemma}[variable-free substitutions are preserved on return]
	Let $S$ be a type specification, $\sigma \in \Sigma$, and $C \subseteq T \times \mathcal{T}$.

	If at least one of the expressions $e_1$ and $e_2$ is variable-free and
	$\sigma$ is variable-free, then every terminating evaluation of
	$\meet_S(e_1,e_2,\sigma,C)$ returns a variable-free substitution.
\end{lemma}

\begin{proof}
	Proof idea: we prove the lemma by induction over the finite call tree depth
	of the terminating evaluation.
\end{proof}

\begin{lemma}[variable-free substitution invariant]
	Let $S$ be a type specification, $\sigma \in \Sigma$, and $C \subseteq T \times \mathcal{T}$.

	If at least one of the expressions $e_1$ and $e_2$ is variable-free
	and $\sigma$ is variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	the substitution $\sigma'$ is variable-free.
\end{lemma}

\begin{proof}
	Proof idea: The meet computation either directly forwards a variable-free
	substitution or uses a variable-free substitution returned by a recursive
	call in subsequent meet computations.
\end{proof}

\begin{lemma}[variable-freeness is absorbing]\label{lem:absorbing}
	Let $\hat e_1,\hat e_2 \in \mathcal{T}$ and $\hat\sigma \in \Sigma$ be variable-free,
	and let $\hat C \subseteq T \times \mathcal{T}$.
	Then in the call tree of $\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$:
	\begin{enumerate}
		\item[(i)] every call $\meet_S(e_1,e_2,\sigma,C)$ has $e_1$, $e_2$, and $\sigma$ variable-free, and
		\item[(ii)] if a call returns $(e',\sigma')$, then $e'$ and $\sigma'$ are variable-free.
	\end{enumerate}
\end{lemma}

\begin{proof}
	(i) and (ii) are proved together by induction on the order in which calls are issued and completed.
	Every appeal is to an earlier point in this order: (i) at a call uses (i) at its parent
	and (ii) at an earlier recursive call, which has returned; (ii) uses (i) at the same call and
	(ii) at its recursive calls.

	\emph{Proof of (i).}
	The root holds by hypothesis. Assume it at a call.
	\begin{itemize}
		\item \textbf{Variable expansion.} Cannot apply, as neither argument is a bare variable.
		\item \textbf{Type unfolding, Subtype unfolding.}
		      The recursive call receives one current argument, which is variable-free,
		      and a definition body, which is variable-free because type specifications contain no variables.
		      The mapping $\sigma$ is unchanged.
		\item \textbf{Union decomposition.}
		      The recursive calls receive subterms of $e_1$ and $e_2$, which are variable-free,
		      and the mapping $\sigma$, unchanged.
		\item \textbf{Function decomposition.}
		      The $i$-th recursive call receives subterms of $e_1$ and $e_2$, which are variable-free.
		      Its mapping $\sigma_{i-1}$ is variable-free:
		      for $i = 1$ it is $\sigma_0 = \sigma$,
		      and for $i > 1$ it is returned by the $(i-1)$-st recursive call, to which (ii) applies.
		\item \textbf{Terminal cases.} No recursive call.
	\end{itemize}

	\emph{Proof of (ii).}
	By (i), variable expansion never applies in the tree,
	so every case returns $\sigma$, the mapping of a recursive call,
	or a join of such mappings, all variable-free
	since $(\sigma_1 \sqcup \sigma_2)(v) = \sigma_1(v) \mid \sigma_2(v)$.
	Hence $\sigma'$ is variable-free, and it remains to show the same for $e'$.
	\begin{itemize}
		\item \textbf{Terminal cases.} The result is an argument, a constant, or $\bot$.
		      Arguments are variable-free by (i), and constants and $\bot$ contain no variables.
		\item \textbf{Type unfolding, Subtype unfolding.}
		      The result is the result of the recursive call, or a constant, or $\bot$.
		      The induction hypothesis applies to the recursive call.
		\item \textbf{Union decomposition.}
		      The result is $e_{i_1}' \mid \dots \mid e_{i_k}'$, where each $e_{i_j}'$ is the result of a recursive call,
		      hence variable-free by the induction hypothesis.
		\item \textbf{Function decomposition.}
		      The result is $f(h_1,\dots,h_n)$ or $\bot$, where each $h_i$ is the result of a recursive call,
		      hence variable-free by the induction hypothesis. \qedhere
	\end{itemize}
\end{proof}

\begin{lemma}[one-sided variable-freeness]\label{lem:onesided}
	Let $\hat e_1,\hat e_2 \in \mathcal{T}$ with $\hat e_1$ or $\hat e_2$ variable-free,
	let $\hat\sigma \in \Sigma$ be variable-free, and let $\hat C \subseteq T \times \mathcal{T}$.
	Then in the call tree of $\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$:
	\begin{enumerate}
		\item[(i)] every call $\meet_S(e_1,e_2,\sigma,C)$ has $\sigma$ variable-free,
		      and $e_1$ or $e_2$ variable-free, and
		\item[(ii)] if a call returns $(e',\sigma')$, then $\sigma'$ is variable-free.
	\end{enumerate}
\end{lemma}

\begin{proof}
	(i) and (ii) are proved together by induction on the order in which calls are issued and completed.
	Every appeal is to an earlier point in this order: (i) at a call uses (i) at its parent
	and (ii) at an earlier recursive call, which has returned; (ii) uses (i) at the same call and
	(ii) at its recursive calls.

	\emph{Proof of (i).} The root satisfies the claim by hypothesis.
	Assume the claim at a call.
	\begin{itemize}
		\item \textbf{Type unfolding.}
		      The recursive call receives the definition body, which is variable-free,
		      together with a current argument.
		      So one of its arguments is variable-free. The mapping $\sigma$ is unchanged.
		\item \textbf{Subtype unfolding.}
		      The recursive call receives the definition body $e$, which is variable-free,
		      together with the current argument $f(e_1,\dots,e_n)$.
		      So one of its arguments is variable-free. The mapping $\sigma$ is unchanged.
		\item \textbf{Variable expansion.}
		      As $v$ is a variable, the hypothesis forces the other argument $e$ to be variable-free,
		      and $\sigma(v)$ is variable-free by hypothesis.
		      Hence both arguments of the recursive call are variable-free,
		      and the mapping it receives is $\sigma$, unchanged.
		\item \textbf{Union decomposition.}
		      The mapping $\sigma$ is unchanged.
		      If $e$ is the variable-free argument, it is again an argument of each recursive call.
		      If the union is the variable-free argument, each part $e_i$ is variable-free.
		\item \textbf{Function decomposition.}
		      The $i$-th recursive call is $\meet_S(e_i,g_i,\sigma_{i-1},C)$.
		      Its arguments $e_i$ and $g_i$ are subterms of $f(e_1,\dots,e_n)$ and $f(g_1,\dots,g_n)$.
		      A subterm of a variable-free term is variable-free,
		      so whichever of the two is variable-free by hypothesis supplies a variable-free argument.
		      Its mapping $\sigma_{i-1}$ is also variable-free:
		      for $i = 1$ we have $\sigma_0 = \sigma$, variable-free by hypothesis,
		      and for $i > 1$ the mapping $\sigma_{i-1}$ is returned by the $(i-1)$-st recursive call,
		      to which (ii) applies.
		\item \textbf{Terminal cases.} No recursive call.
	\end{itemize}

	\emph{Proof of (ii).} The only case that alters the mapping is variable expansion, which returns
	$\{v' \mapsto \sigma'(v') \mid v' \neq v \in V\} \cup \{v \mapsto e'\}$,
	where $(e',\sigma')$ is the result of the recursive call.
	By (i) that recursive call has both arguments and its mapping variable-free,
	so by Lemma~\ref{lem:absorbing}(ii) both $e'$ and $\sigma'$ are variable-free,
	and hence so is the returned mapping.
	In every other case the returned mapping is the mapping of a recursive call
	(type unfolding, subtype unfolding, function decomposition),
	a join $\sigma_{i_1} \sqcup \dots \sqcup \sigma_{i_k}$ of such mappings (union decomposition),
	or $\sigma$ itself (terminal cases).
	In the first case the induction hypothesis applies.
	In the second it applies to each $\sigma_{i_j}$, and a join of variable-free mappings is variable-free,
	since $(\sigma_1 \sqcup \sigma_2)(v) = \sigma_1(v) \mid \sigma_2(v)$.
	In the third the mapping is variable-free by (i).
\end{proof}

Note that (ii) makes no claim about $e'$, and cannot:
the \emph{top} case returns the other argument unchanged,
so a call satisfying the hypothesis may well return an expression containing variables.

\begin{lemma}[argument pool]\label{lem:pool}
	Let $\hat e_1,\hat e_2 \in \mathcal{T}$, $\hat\sigma \in \Sigma$, and $\hat C \subseteq T \times \mathcal{T}$.
	Consider a sequence of calls along a branch starting at
	$\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$ in which no var-step occurs.
	Then every argument of every call in that sequence lies in
	\[
		P = \subexp(\hat e_1) \cup \subexp(\hat e_2) \cup S_{\mathrm{sub}}.
	\]
\end{lemma}

\begin{proof}
	By induction along the sequence. The first arguments lie in $P$. Assume a call's arguments lie in $P$,
	and consider its successor in the sequence.
	\begin{itemize}
		\item \textbf{Variable expansion.} Excluded by hypothesis.
		\item \textbf{Type unfolding, Subtype unfolding.}
		      The successor call receives one current argument, which lies in $P$,
		      and a definition body, which lies in $S_{\mathrm{sub}} \subseteq P$.
		\item \textbf{Union decomposition.}
		      The successor receives $e_i$, a part of a current argument and hence a subterm of it,
		      and $e$, a current argument. Both lie in $P$.
		\item \textbf{Function decomposition.}
		      The successor receives $e_i$ and $g_i$, subterms of current arguments, hence in $P$.
		\item \textbf{Terminal cases.} No recursive call. \qedhere
	\end{itemize}
\end{proof}

\begin{lemma}[at most one variable expansion]\label{lem:onevar}
	Let $\hat e_1,\hat e_2 \in \mathcal{T}$ with $\hat e_1$ or $\hat e_2$ variable-free,
	let $\hat\sigma \in \Sigma$ be variable-free, and let $\hat C \subseteq T \times \mathcal{T}$.
	Then every branch of the call tree of $\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$ contains at most one var-step.
\end{lemma}

\begin{proof}
	A var-step requires an argument that is a bare variable.
	By Lemma~\ref{lem:onesided}(i), at such a call the mapping $\sigma$ is variable-free
	and the other argument is variable-free.
	Hence the recursive call has both arguments and its mapping variable-free.
	By Lemma~\ref{lem:absorbing}(i), applied to the subtree rooted at that call,
	every call below it has variable-free arguments,
	so no bare variable occurs and variable expansion cannot apply again.
\end{proof}

\begin{lemma}[branches are finite]\label{lem:finitebranch}
	Let $\hat e_1,\hat e_2 \in \mathcal{T}$ with $\hat e_1$ or $\hat e_2$ variable-free,
	let $\hat\sigma \in \Sigma$ be variable-free, and let $\hat C \subseteq T \times \mathcal{T}$.
	Then every branch of the call tree of $\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$ is finite.
\end{lemma}

\begin{proof}
	By Lemma~\ref{lem:onevar} the branch splits at its var-step into at most two \emph{segments},
	neither containing a var-step.
	Fix a segment and let $\meet_S(a_1,a_2,\sigma_a,C_a)$ be its first call.
	Since the segment is a sequence of calls along a branch in which no var-step occurs,
	Lemma~\ref{lem:pool} applies, and every argument in the segment lies in the finite set
	\[
		P = \subexp(a_1) \cup \subexp(a_2) \cup S_{\mathrm{sub}}.
	\]
	Every pair added to $C$ arises from an unfolding. Its first component is the unfolded constant,
	which is defined in $S$, and its second is an argument.
	So every pair added within the segment lies in $T_S \times P$, a finite set.
	For a call $\meet_S(e_1,e_2,\sigma,C)$ of the segment, consider the measure
	\[
		M = \Big(\ \big|(T_S \times P) \setminus C\big| \ ,\ \ \mathrm{size}(e_1) + \mathrm{size}(e_2) \ \Big) \in \mathbb{N}^2,
	\]
	ordered lexicographically.
	\begin{itemize}
		\item \textbf{Type unfolding.}
		      Adds the pair $(t,e) \notin C$ to $C$, so the first component of $M$ strictly decreases.
		\item \textbf{Subtype unfolding.}
		      Adds the pair $(t,f(e_1,\dots,e_n)) \notin C$ to $C$, so the first component of $M$ strictly decreases.
		\item \textbf{Variable expansion.} Does not occur in a segment.
		\item \textbf{Union decomposition.}
		      The guard is unchanged, and the recursive call receives $e_i$, a strict subterm of $e_1 \mid \dots \mid e_n$,
		      together with $e$. Note the two arguments are exchanged.
		      The sum of argument sizes strictly decreases, so the second component of $M$ strictly decreases.
		\item \textbf{Function decomposition.}
		      The guard is unchanged, and the recursive call receives $e_i$ and $g_i$,
		      strict subterms of $f(e_1,\dots,e_n)$ and $f(g_1,\dots,g_n)$.
		      The sum of argument sizes strictly decreases, so the second component of $M$ strictly decreases.
		\item \textbf{Terminal cases.} No recursive call, and the branch ends.
	\end{itemize}
	The lexicographic order on $\mathbb{N}^2$ is well-founded and $M$ strictly decreases at every step,
	so the segment is finite.
	A branch consists of at most two finite segments, hence is finite.
\end{proof}

\subsection*{Termination}

\begin{theorem}\label{thm:termination}
	Let $\hat e_1,\hat e_2 \in \mathcal{T}$ with $\hat e_1$ or $\hat e_2$ variable-free,
	let $\hat\sigma \in \Sigma$ be variable-free, and let $\hat C \subseteq T \times \mathcal{T}$.
	Then $\meet_S(\hat e_1,\hat e_2,\hat\sigma,\hat C)$ terminates.
\end{theorem}

\begin{proof}
	Every case makes finitely many recursive calls, so the call tree is finitely branching.
	By Lemma~\ref{lem:finitebranch} it has no infinite branch.
	By K\"onig's lemma a finitely branching tree with no infinite branch is finite,
	hence the call tree is finite.
\end{proof}

\begin{corollary}[type checking terminates]
	In typclingo the meet is invoked repeatedly, each call taking as second argument
	a type expression built from $S$, hence variable-free,
	and as mapping the one returned by the previous call,
	starting from the initial mapping, which sends every variable to $\mathbb{S}$
	and is therefore variable-free.
	By Lemma~\ref{lem:onesided}(ii) each returned mapping is again variable-free,
	so every call satisfies the hypothesis of Theorem~\ref{thm:termination} and terminates.
\end{corollary}
