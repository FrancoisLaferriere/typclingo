\section*{Termination of the meet computation}

\todoin{\textbf{Proof overview.}
	The main difficulty is to show that the arguments are drawn from a finite set.
	This is clear if no variable expansion occurs:
	every recursive call then receives either a subterm of a current argument
	or a definition body from $S$.
	In this case, the call guard set prevents infinite unfolding of recursive types.
	Variable expansion is the only case that introduces expressions from outside this set.
	Assuming that at least one of the initial type-expression arguments
	and the initial mapping are variable-free,
	at most one variable expansion can occur along branches.
	After that expansion, both arguments are variable-free and remain so.
	Hence termination reduces to the variable-free case.}

\subsection*{Definitions}

For a type expression $e$, the set $\subexp(e)$ of its \emph{subterms} is defined by
\[
	\subexp(e) = \{e\} \cup
	\begin{cases}
		\bigcup_{i=1}^n \subexp(e_i) & \text{if } e = f(e_1,\dots,e_n) \text{ or } e = e_1 \mid \dots \mid e_n, \\
		\emptyset                    & \text{otherwise.}
	\end{cases}
\]

For a type specification $S$, let $\subexp(S) = \bigcup_{(t \lessdot e)\in S\,\text{or}\,(t \doteq e)\in S} \subexp(e)$
be the set of subterms of all definition bodies. It is finite.

Let $T_S = \{\, t : (t \lessdot e) \in S \text{ or } (t \doteq e) \in S \,\}$
be the constants defined in $S$. $T_S$ is finite, since $S$ is.

For a type expression $e$, its \emph{size} is defined by
\[
	\mathrm{size}(e) =
	\begin{cases}
		1 + \sum_{i=1}^n \mathrm{size}(e_i) & \text{if } e = f(e_1,\dots,e_n) \text{ or } e = e_1 \mid \dots \mid e_n, \\
		1                               & \text{otherwise.}
	\end{cases}
\]

An expression is \emph{variable-free} if no type variable occurs in it.
A mapping $\sigma \in \Sigma$ is \emph{variable-free} if $\sigma(v)$ is variable-free for every $v \in V$.

The \emph{call tree} of $\meet_S(e_1, e_2, \sigma, C)$ has that call as its root,
and at each node the recursive calls made by the applicable case as its children.
We identify termination of $\meet_S$ with finiteness of its call tree.

\subsection*{Lemmas}

\begin{lemma}[variable-freeness invariant]\label{lem:vfree-inv}
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	and $C \subseteq T \times \mathcal{T}$.

	If type expressions $e_1$ and $e_2$ are variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	both $e_1'$ and $e_2'$ are variable-free.
\end{lemma}

\begin{proof}
	By inspection of the definition of $\meet_S$,
	variable-freeness of the first two arguments is preserved under direct recursive subcalls.
	Since the initial call $\meet_S(e_1,e_2,\sigma,C)$ has variable-free
	first two arguments, it follows that every reachable subcall
	$\meet_S(e_1',e_2',\sigma',C')$ also has variable-free first two
	arguments.
\end{proof}

If the type expressions in all recursive calls are variable-free,
we obtain the following corollary.

\begin{corollary}[no variable expansion]\label{cor:novarexp}
	There is no variable expansion step during the evaluation of a meet with
	two variable-free type expressions.
\end{corollary}

The next invariant shows that we can restrict attention to a finite set of type
expressions when analyzing the meet computation.

\begin{lemma}[variable-free argument pool]\label{lem:vfree-pool}
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	$C \subseteq T \times \mathcal{T}$,
	and $P = \subexp(e_1) \cup \subexp(e_2) \cup \subexp(S)$.

	If type expressions $e_1$ and $e_2$ are variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	$e_1',e_2' \in P$.
\end{lemma}

\begin{proof}
	By Corollary~\ref{cor:novarexp} there is no variable expansion step during the
	evaluation of a meet with two variable-free type expressions, so every subcall
	receives either a subterm of one of the current arguments or a definition body from
	$S$. Therefore, all arguments of all reachable subcalls are contained in the finite set $P$.
\end{proof}

\begin{lemma}[variable-free termination]\label{lem:vfree-term}
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	and $C \subseteq T \times \mathcal{T}$.

	If type expressions $e_1$ and $e_2$ are variable-free,
	then the meet computation $\meet_S(e_1,e_2,\sigma,C)$ terminates.
\end{lemma}

\begin{proof}
	By Lemma~\ref{lem:vfree-pool} every argument of every reachable call lies in the finite set
    \[
        P = \subexp(e_1) \cup \subexp(e_2) \cup \subexp(S),
    \]
	so every guard pair added during the evaluation lies in the finite set $T_S \times P$.
	For a call $\meet_S(e_1',e_2',\sigma',C')$ consider the measure
	\[
		M = \Big(\ \big|(T_S \times P) \setminus C'\big| \ ,\ \ \mathrm{size}(e_1') + \mathrm{size}(e_2') \ \Big) \in \mathbb{N}^2,
	\]
	ordered lexicographically.
	By Corollary~\ref{cor:novarexp} no variable expansion occurs.
	Each recursive call either adds a pair to $C$ that was not already present,
	strictly decreasing the first component of $M$,
	or leaves $C$ unchanged and receives strict subterms of the current arguments,
	strictly decreasing the second component.
	Thus $M$ strictly decreases from each call to every call it makes,
	and the lexicographic order on $\mathbb{N}^2$ is well-founded, so the call tree has no infinite path.
	Since each case makes finitely many recursive calls, the tree is finitely branching,
	hence finite by K\"onig's lemma, and the evaluation terminates.
\end{proof}

\begin{lemma}[one-sided variable-freeness invariant]\label{lem:onesided}
	Let $S$ be a type specification,
	$\sigma \in \Sigma$,
	and $C \subseteq T \times \mathcal{T}$.

	If at least one of the expressions $e_1$ and $e_2$ is variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	at least one of the expressions $e_1'$ and $e_2'$ is variable-free.
\end{lemma}
\begin{proof}
	It suffices to check that each recursive call preserves having a variable-free argument.
	This is immediate except in variable expansion, $\meet_S(v,e,\sigma,C)$, where $v$ is a bare
	variable, so the hypothesis forces the other argument $e$ to be variable-free.
	The recursive call $\meet_S(\sigma(v),e,\sigma,C)$ then still has the variable-free argument $e$.
\end{proof}

\begin{lemma}[variable-free results]\label{lem:vfree-result}
	Let $S$ be a type specification, $\sigma \in \Sigma$ variable-free,
	and $C \subseteq T \times \mathcal{T}$, with $e_1$ and $e_2$ variable-free.
	Then every terminating evaluation of $\meet_S(e_1,e_2,\sigma,C)$
	returns a variable-free expression and a variable-free substitution.
\end{lemma}
\begin{proof}
	By Corollary~\ref{cor:novarexp} variable expansion does not occur, so the substitution
	is never altered and the returned substitution is $\sigma$, variable-free by hypothesis.
	For the expression, argue by induction on the height of the finite call tree.
	By Lemma~\ref{lem:vfree-inv} every reachable call has variable-free arguments,
	so the induction hypothesis applies to each recursive call.
	The returned expression is an argument, a constant, $\bot$, or an expression built from
	the results of recursive calls. Arguments are variable-free by Lemma~\ref{lem:vfree-inv},
	the others by the induction hypothesis.
\end{proof}

\begin{lemma}[variable-free substitutions are preserved on return]\label{lem:subreturn}
	Let $S$ be a type specification, $\sigma \in \Sigma$, and $C \subseteq T \times \mathcal{T}$.

	If at least one of the expressions $e_1$ and $e_2$ is variable-free and
	$\sigma$ is variable-free, then every terminating evaluation of
	$\meet_S(e_1,e_2,\sigma,C)$ returns a variable-free substitution.
\end{lemma}

\begin{proof}
	We prove the lemma by induction over the finite call tree depth
	of the terminating evaluation.
	The substitution is altered only by variable expansion, at $\meet_S(v,e,\sigma,C)$,
	which returns $\{v' \mapsto \sigma'(v') : v' \neq v\} \cup \{v \mapsto e'\}$,
	where $(e',\sigma')$ is the result of the recursive call $\meet_S(\sigma(v),e,\sigma,C)$.
	By Lemma~\ref{lem:onesided} the argument $e$ is variable-free, and $\sigma(v)$ is variable-free
	since $\sigma$ is, so that call has both arguments variable-free and terminates by
	Lemma~\ref{lem:vfree-term}. By Lemma~\ref{lem:vfree-result} its results $e'$ and $\sigma'$ are variable-free,
	so the returned substitution is variable-free.
	Every other case returns $\sigma$ or a substitution returned by a recursive call,
	or a join of such, all variable-free by the induction hypothesis.
\end{proof}

\begin{lemma}[variable-free substitution invariant]\label{lem:subinv}
	Let $S$ be a type specification, $\sigma \in \Sigma$, and $C \subseteq T \times \mathcal{T}$.

	If at least one of the expressions $e_1$ and $e_2$ is variable-free
	and $\sigma$ is variable-free,
	then for every subcall $\meet_S(e_1',e_2',\sigma',C')$
	reachable during the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	the substitution $\sigma'$ is variable-free.
\end{lemma}

\begin{proof}
	By induction on the depth of the reachable calls. The root receives $\sigma$, variable-free by hypothesis.
	Suppose a reachable call receives a variable-free mapping and consider the mappings its recursive calls receive.
	In every case other than function decomposition each recursive call receives the current mapping unchanged,
	which is variable-free.
	In function decomposition the $i$-th recursive call $\meet_S(e_i,g_i,\sigma_{i-1},C')$ receives $\sigma_{i-1}$.
	We show by induction on $i$ that each $\sigma_{i-1}$ is variable-free.
	For $i=1$ it is the current mapping, variable-free.
	For $i>1$ it is the mapping returned by the $(i-1)$-st recursive call.
	Since the $i$-th call is reachable, the $(i-1)$-st has returned.
	That call receives $\sigma_{i-2}$, variable-free by the inner induction hypothesis,
	and has a variable-free argument by Lemma~\ref{lem:onesided},
	so by Lemma~\ref{lem:subreturn} its returned mapping $\sigma_{i-1}$ is variable-free.
\end{proof}

\begin{lemma}[at most one variable expansion per branch]\label{lem:onevar}
	Let $S$ be a type specification, $\sigma \in \Sigma$ variable-free,
	and $C \subseteq T \times \mathcal{T}$ with at least one of $e_1$ and $e_2$ variable-free,
	then every branch of the evaluation of $\meet_S(e_1,e_2,\sigma,C)$
	contains at most one variable expansion step.
\end{lemma}

\begin{proof}
	Suppose a branch has a variable expansion step $\meet_S(v,e,\sigma',C')$.
	By Lemma~\ref{lem:onesided} one of its arguments is variable-free,
	and since $v$ is a bare variable, $e$ is variable-free.
	By Lemma~\ref{lem:subinv} $\sigma'$ is variable-free, so $\sigma'(v)$ is variable-free.
	Hence the successor call $\meet_S(\sigma'(v),e,\sigma',C')$ has both arguments variable-free,
	and by Lemma~\ref{lem:vfree-inv} every call in the subtree below it
	has both arguments variable-free. No bare variable occurs there,
	so no further variable expansion step is possible on the branch.
\end{proof}

\begin{lemma}[argument pool along a branch]\label{lem:branchpool}
	Let $S$ be a type specification, $\sigma \in \Sigma$ variable-free,
	and $C \subseteq T \times \mathcal{T}$,
	with at least one of $e_1$ and $e_2$ variable-free.
	Along a branch of the evaluation of $\meet_S(e_1,e_2,\sigma,C)$,
	let $E = \subexp(\sigma'(v))$ if the branch has a variable expansion step $\meet_S(v,e,\sigma',C')$,
	and $E = \emptyset$ otherwise.
	Then every argument of every call along the branch lies in the finite set
	\[
		P = \subexp(e_1) \cup \subexp(e_2) \cup \subexp(S) \cup E.
	\]
\end{lemma}

\begin{proof}
	$P$ is finite: by Lemma~\ref{lem:onevar} the branch has at most one variable expansion step,
	so $E$ is $\subexp$ of at most one expression.
	Argue by induction along the branch. The first arguments $e_1, e_2$ lie in $P$.
	For the induction step, every case passes to its successor a subterm of a current
	argument or a definition body, both in $P$, except variable expansion,
	which passes $\sigma'(v) \in E \subseteq P$ alongside the current argument $e$.
\end{proof}

\begin{lemma}[branches are finite]\label{lem:finitebranch}
	Let $S$ be a type specification, $\sigma \in \Sigma$ variable-free,
	and $C \subseteq T \times \mathcal{T}$,
	with at least one of $e_1$ and $e_2$ variable-free.
	Then every branch of the evaluation of $\meet_S(e_1,e_2,\sigma,C)$ is finite.
\end{lemma}
\begin{proof}
	Fix a branch. By Lemma~\ref{lem:branchpool} every argument along it lies in a finite set $P$,
	so every pair added to $C$ along the branch lies in the finite set $T_S \times P$.
	For a call $\meet_S(e_1',e_2',\sigma',C')$ of the branch, let $b \in \{0,1\}$ be $1$
	if a variable occurs in $e_1'$ or $e_2'$, and $0$ otherwise. Consider the measure
	\[
		M = \Big(\ \big|(T_S \times P) \setminus C'\big| \ ,\ \ b \ ,\ \ \mathrm{size}(e_1') + \mathrm{size}(e_2') \ \Big) \in \mathbb{N}^3,
	\]
	ordered lexicographically.
	Each step decreases $M$. A step that adds a pair to $C$ not already present
	strictly decreases the first component. A variable expansion leaves $C$ unchanged
	and replaces the bare variable by the variable-free expression $\sigma'(v)$
	while the other argument is variable-free, so no variable remains and $b$ drops from $1$ to $0$,
	strictly decreasing the second component.
	Any other step either ends the branch, making no recursive call,
	or leaves $C$ and $b$ unchanged and passes strict subterms of the current arguments
	to the successor, strictly decreasing the third component.
	The lexicographic order on $\mathbb{N}^3$ is well-founded and $M$ strictly decreases at every step,
	so the branch is finite.
\end{proof}

\subsection*{Termination}

\begin{theorem}\label{thm:termination}
	Let $S$ be a type specification, $\sigma \in \Sigma$ variable-free,
	and $C \subseteq T \times \mathcal{T}$,
	with at least one of $e_1$ and $e_2$ variable-free.
	Then $\meet_S(e_1,e_2,\sigma,C)$ terminates.
\end{theorem}

\begin{proof}
	Every case makes finitely many recursive calls, so the call tree is finitely branching.
	By Lemma~\ref{lem:finitebranch} it has no infinite branch.
	By K\"onig's lemma a finitely branching tree with no infinite branch is finite,
	hence the call tree is finite.
\end{proof}

\begin{corollary}[type checking terminates]
	In typclingo the meet is invoked repeatedly. Each call takes as second argument a
	type expression built from $S$, which is variable-free, and as mapping the one returned
	by the previous call. The first call starts from the initial mapping, which sends every
	variable to $\mathbb{S}$ and is variable-free. Then every call terminates.
\end{corollary}

\begin{proof}
	By induction over the sequence of calls.
	The first call has a variable-free second argument and a variable-free mapping.
	Suppose some call does. By Theorem~\ref{thm:termination} it terminates,
	and by Lemma~\ref{lem:subreturn} it returns a variable-free mapping.
	The next call receives that mapping and again has a variable-free second argument,
	so the claim holds for it.
\end{proof}
